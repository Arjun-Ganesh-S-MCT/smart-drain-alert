Serial format and wiring help
   

   
Send one JSON line per second at 115200 baud: {"flow":4.2,"dist":12.5,"turb":850,"rain":0}
flow = L/min, dist = cm to water, turb = higher means dirtier, rain = 1 wet / 0 dry.
Arduino: Serial.printf("{\"flow\":%.2f,\"dist\":%.1f,\"turb\":%d,\"rain\":%d}\n",flow,dist,turb,rain);

   
Wiring reminders: the ESP32-S3 inputs accept 3.3 V only. Put the 4-channel voltage divider on the ultrasonic echo, the turbidity analog output and the flow signal before connecting. Confirm the rain module pin labels first. Join all grounds. Keep the ultrasonic control board and turbidity module dry. Test each sensor alone, then combine.

  
ESP32-S3 firmware for Wi-Fi and SMS

SMS needs a GSM module (for example SIM800L) with a SIM card on the ESP32-S3, because a PC cannot send SMS by itself. The app sends commands as JSON lines. Start Serial1 on your GSM module pins in setup(), and for every reading print the JSON line and also call ws.broadcastTXT(json) once Wi-Fi is on.


#include <WiFi.h>
#include <WebSocketsServer.h>
#include <ArduinoJson.h>
WebSocketsServer ws(81);
bool wsOn=false;
void sendSms(const String& to,const String& msg){
  Serial1.println("AT+CMGF=1");delay(300);
  Serial1.println("AT+CMGS=\""+to+"\"");delay(300);
  Serial1.print(msg);Serial1.write(26);
}
void handle(const String& line){
  JsonDocument d;if(deserializeJson(d,line))return;
  String c=d["cmd"];
  if(c=="sms")sendSms(d["to"].as<String>(),d["msg"].as<String>());
  if(c=="wifi")WiFi.begin(d["ssid"].as<const char*>(),d["pass"].as<const char*>());
}
void loop(){
  if(Serial.available())handle(Serial.readStringUntil('\n'));
  if(WiFi.status()==WL_CONNECTED&&!wsOn){ws.begin();wsOn=true;Serial.printf("{\"ip\":\"%s\"}\n",WiFi.localIP().toString().c_str());}
  if(wsOn)ws.loop();
}
